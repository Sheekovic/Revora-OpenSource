cmake_minimum_required(VERSION 3.25)
project(RevoraAuthClient LANGUAGES CXX)
add_library(RevoraAuthClient SHARED src/revora_auth_client.cpp src/direct_native_runtime.cpp src/owned_login_controls.cpp src/revora_auth_client.def)
target_compile_features(RevoraAuthClient PRIVATE cxx_std_20)
target_compile_options(RevoraAuthClient PRIVATE /W4 /WX /guard:cf /sdl)
target_link_options(RevoraAuthClient PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
target_link_libraries(RevoraAuthClient PRIVATE winhttp bcrypt crypt32 ole32 comctl32 ws2_32)
set_target_properties(RevoraAuthClient PROPERTIES OUTPUT_NAME RevoraAuthClient)
option(REVORA_GAME_HANDSHAKE_FIXTURE "Research-only fixed loopback Game handshake boundary" OFF)
if(REVORA_GAME_HANDSHAKE_FIXTURE)
  target_compile_definitions(RevoraAuthClient PRIVATE REVORA_GAME_HANDSHAKE_FIXTURE=1)
endif()
add_executable(RevoraAuthClientTests tests/revora_auth_client_tests.cpp)
add_executable(RevoraGameSendTransparencyTests tests/game_send_transparency_tests.cpp)
target_compile_features(RevoraGameSendTransparencyTests PRIVATE cxx_std_20)
target_compile_options(RevoraGameSendTransparencyTests PRIVATE /W4 /WX /sdl)
target_link_libraries(RevoraGameSendTransparencyTests PRIVATE RevoraAuthClient ws2_32)
add_executable(RevoraWorldHandoffTests tests/world_handoff_tests.cpp)
target_compile_features(RevoraWorldHandoffTests PRIVATE cxx_std_20)
target_compile_options(RevoraWorldHandoffTests PRIVATE /W4 /WX /sdl)
add_executable(RevoraWorldStubTests tests/world_stub_tests.cpp)
target_compile_features(RevoraWorldStubTests PRIVATE cxx_std_20)
target_compile_options(RevoraWorldStubTests PRIVATE /W4 /WX /sdl)
target_compile_features(RevoraAuthClientTests PRIVATE cxx_std_20)
target_compile_options(RevoraAuthClientTests PRIVATE /W4 /WX)
target_link_libraries(RevoraAuthClientTests PRIVATE RevoraAuthClient ole32)
add_executable(RevoraOwnedLoginControlsTests tests/owned_login_controls_tests.cpp src/owned_login_controls.cpp)
target_compile_features(RevoraOwnedLoginControlsTests PRIVATE cxx_std_20)
target_compile_options(RevoraOwnedLoginControlsTests PRIVATE /W4 /WX /sdl)
target_link_libraries(RevoraOwnedLoginControlsTests PRIVATE comctl32)
