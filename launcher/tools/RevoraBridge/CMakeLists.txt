cmake_minimum_required(VERSION 3.24)
project(RevoraBridge LANGUAGES CXX)

if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "RevoraBridge proof must be built for Windows x86")
endif()

add_library(RevoraBridge SHARED src/bridge.cpp src/bridge.def)
target_compile_features(RevoraBridge PRIVATE cxx_std_20)
target_compile_definitions(RevoraBridge PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(RevoraBridge PRIVATE /W4 /WX /guard:cf /sdl)
target_link_options(RevoraBridge PRIVATE /WX /DYNAMICBASE /NXCOMPAT /GUARD:CF)
set_target_properties(RevoraBridge PROPERTIES OUTPUT_NAME "RevoraNativeLogin" PREFIX "")
target_link_libraries(RevoraBridge PRIVATE bcrypt winhttp ole32 ws2_32)
